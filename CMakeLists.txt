cmake_minimum_required(VERSION 3.16)
project(big_red_button LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# On Unix: auto (GTK3 when available, otherwise raw X11), gtk3, or x11.
set(BUTTON_BACKEND auto CACHE STRING "Unix window backend: auto, gtk3, x11")
set_property(CACHE BUTTON_BACKEND PROPERTY STRINGS auto gtk3 x11)

if(WIN32)
    add_executable(redbutton WIN32 src/win32.cpp)
    target_link_libraries(redbutton PRIVATE user32 gdi32)
    target_compile_definitions(redbutton PRIVATE UNICODE _UNICODE)
elseif(APPLE)
    enable_language(OBJCXX)
    add_executable(redbutton MACOSX_BUNDLE src/macos.mm)
    target_compile_options(redbutton PRIVATE -fobjc-arc)
    target_link_libraries(redbutton PRIVATE "-framework Cocoa" "-framework QuartzCore")
    set_target_properties(redbutton PROPERTIES
        MACOSX_BUNDLE_BUNDLE_NAME "Big Red Button"
        MACOSX_BUNDLE_GUI_IDENTIFIER "org.example.bigredbutton")
elseif(CMAKE_SYSTEM_NAME STREQUAL "Haiku")
    add_executable(redbutton src/haiku.cpp)
    target_link_libraries(redbutton PRIVATE be)
elseif(UNIX)
    if(NOT BUTTON_BACKEND MATCHES "^(auto|gtk3|x11)$")
        message(FATAL_ERROR "BUTTON_BACKEND must be auto, gtk3, or x11")
    endif()

    find_package(PkgConfig QUIET)
    if(PkgConfig_FOUND AND NOT BUTTON_BACKEND STREQUAL "x11")
        pkg_check_modules(GTK3 QUIET IMPORTED_TARGET gtk+-3.0)
    endif()

    if(BUTTON_BACKEND STREQUAL "gtk3" AND NOT GTK3_FOUND)
        message(FATAL_ERROR "GTK3 requested, but gtk+-3.0 was not found")
    endif()

    if(GTK3_FOUND AND NOT BUTTON_BACKEND STREQUAL "x11")
        add_executable(redbutton src/gtk3.cpp)
        target_link_libraries(redbutton PRIVATE PkgConfig::GTK3)
        message(STATUS "Red button backend: GTK3 (Wayland or X11)")
    else()
        find_package(X11 REQUIRED)
        add_executable(redbutton src/x11.cpp)
        target_link_libraries(redbutton PRIVATE X11::X11)
        message(STATUS "Red button backend: Xlib (X11)")
    endif()
else()
    message(FATAL_ERROR "No backend implemented for ${CMAKE_SYSTEM_NAME}")
endif()

if(MSVC)
    target_compile_options(redbutton PRIVATE /W4 /permissive-)
else()
    target_compile_options(redbutton PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(test_reasons tests/test_reasons.cpp)
    add_test(NAME reasons COMMAND test_reasons)
endif()
